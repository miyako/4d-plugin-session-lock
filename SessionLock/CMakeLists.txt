cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS deployment target")
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" PLUGIN_VERSION LIMIT_COUNT 1)
project(SessionLock VERSION ${PLUGIN_VERSION} LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(Threads REQUIRED)
include(CTest)
add_library(sessionlock_core STATIC core.cpp monitor.cpp serialization.cpp)
set_target_properties(sessionlock_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(sessionlock_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(sessionlock_core PUBLIC Threads::Threads)

if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 17)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    add_library(sessionlock_backend STATIC backend_macos.mm)
    target_compile_options(sessionlock_backend PRIVATE -fobjc-arc)
    target_link_libraries(sessionlock_backend PUBLIC "-framework Foundation" "-framework CoreGraphics")
elseif(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR MSVC_CXX_ARCHITECTURE_ID STREQUAL "ARM64")
        message(FATAL_ERROR "Only Windows x64 is supported")
    endif()
    add_library(sessionlock_backend STATIC backend_windows.cpp)
    target_compile_definitions(sessionlock_backend PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00 NOMINMAX)
    target_link_libraries(sessionlock_backend PUBLIC Wtsapi32 User32)
else()
    message(FATAL_ERROR "Only macOS and Windows are supported")
endif()
target_link_libraries(sessionlock_backend PUBLIC sessionlock_core)

set(SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../4D-Plugin-SDK/4D Plugin API")
set(SDK_SOURCE "${SDK_DIR}/4DPluginAPI.c")
set(PLUGIN_RESOURCES manifest.json "${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE"
    "${SDK_DIR}/../LICENSE.md")
set_source_files_properties("${SDK_SOURCE}" PROPERTIES LANGUAGE CXX)
add_library(SessionLock MODULE SessionLock-4dplugin.cpp "${SDK_SOURCE}" ${PLUGIN_RESOURCES})
target_include_directories(SessionLock PRIVATE "${SDK_DIR}")
target_compile_definitions(SessionLock PRIVATE UNICODE _UNICODE)
target_link_libraries(SessionLock PRIVATE sessionlock_backend)
set(PLUGIN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/SessionLock-test/Plugins")
if(APPLE)
    set_source_files_properties("${SDK_SOURCE}" PROPERTIES COMPILE_OPTIONS "-Wno-multichar;-Wno-deprecated-declarations;-Wno-null-conversion")
    set_source_files_properties(${PLUGIN_RESOURCES} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(SessionLock PROPERTIES
        BUNDLE TRUE BUNDLE_EXTENSION bundle PREFIX ""
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
        MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
        LIBRARY_OUTPUT_DIRECTORY "${PLUGIN_DIR}")
    foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(SessionLock PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${CONFIG} "${PLUGIN_DIR}")
    endforeach()
else()
    target_link_options(SessionLock PRIVATE "/DEF:${SDK_DIR}/4DPluginAPI.def")
    set(OUTPUT_DIR "${PLUGIN_DIR}/SessionLock/Contents/Windows64")
    set_target_properties(SessionLock PROPERTIES PREFIX "" SUFFIX ".4DX"
        LIBRARY_OUTPUT_DIRECTORY "${OUTPUT_DIR}" RUNTIME_OUTPUT_DIRECTORY "${OUTPUT_DIR}")
    foreach(CONFIG DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
        set_target_properties(SessionLock PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${CONFIG} "${OUTPUT_DIR}"
            LIBRARY_OUTPUT_DIRECTORY_${CONFIG} "${OUTPUT_DIR}")
    endforeach()
    add_custom_command(TARGET SessionLock POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "${PLUGIN_DIR}/SessionLock/Contents/Resources"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${PLUGIN_RESOURCES}
            "${PLUGIN_DIR}/SessionLock/Contents/Resources"
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(BUILD_TESTING)
    add_executable(core_tests tests/core_tests.cpp)
    target_link_libraries(core_tests PRIVATE sessionlock_core)
    add_test(NAME core COMMAND core_tests)
    add_executable(backend_smoke tests/backend_smoke.cpp)
    target_link_libraries(backend_smoke PRIVATE sessionlock_backend)
    add_test(NAME backend_smoke COMMAND backend_smoke)
    set_tests_properties(core backend_smoke PROPERTIES TIMEOUT 30)
endif()
